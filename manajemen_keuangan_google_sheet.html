Penjabaran & Pemahaman Kebutuhan

Analisis dari gambar spreadsheet & alur kerja yang diinginkan

1. Pemahaman Terhadap Gambar Spreadsheet

  • TABUNGAN (Pos Aset Hijau): Terdiri dari 9 rekening bank & dompet digital (Jenius, BRI, Seabank, Blue BCA, BCA, Jagoo, Neobank, Shopeepay, Gopay) dengan akumulasi awal Rp 25.312.570.
  • HUTANG MODAL (Pos Kewajiban Merah): Terdiri dari 3 sumber pembiayaan/paylater (Shopee Paylater, SpPinjam, Tiktok Paylater) dengan total saldo pinjaman Rp 5.400.000.
  • TOTAL ASET LIQUID (Kuning): Sebesar Rp 30.712.570, didapat dari penjumlahan dana tabungan dan modal pinjaman yang saat ini aktif/dapat diputar.

2. Yang Anda Inginkan dalam Aplikasi

  • Form Input Debit / Kredit: Pengguna dapat mencatat mutasi keluar (kredit) maupun masuk (debit) secara cepat.
  • Pilihan Rekening: Dropdown otomatis berisi 12 rekening/pos sesuai gambar untuk memilih rekening mana yang terpengaruh transaksi.
  • Penyimpanan ke Google Sheet: Data transaksi dan pembaruan saldo dikirimkan secara langsung ke Google Spreadsheet Anda via integrasi Google Apps Script (Web App).

3. Cara Kerja Logika Debit & Kredit pada Aplikasi Ini

• Untuk TABUNGAN: Debit menambah saldo tabungan, sedangkan Kredit mengurangi saldo tabungan.
• Untuk HUTANG MODAL: Debit adalah pelunasan cicilan (mengurangi sisa hutang), sedangkan Kredit adalah penarikan pinjaman baru (menambah hutang).

Integrasi Google Sheets

Hubungkan aplikasi ini langsung ke Google Sheet milik Anda

URL ini akan dipakai untuk mengirim transaksi dan menarik saldo dari sheet "Ringkasan_Saldo".

🚀 Panduan 3 Langkah Memasang di Google Sheet Anda:

  1. Buka file Google Sheets Anda, lalu klik menu Extensions (Ekstensi) > Apps Script.
  2. Hapus kode bawaan, lalu salin dan tempelkan kode di bawah ini ke editor Apps Script.
  3. Klik Deploy (Terapkan) > New Deployment (Penerapan Baru) > pilih jenis Web App. Set "Who has access" ke Anyone (Siapa saja), lalu salin Web App URL-nya dan tempel ke kotak di atas!
Kode Google Apps Script (Code.gs): 
/**
 * FinTrack Sheets - Google Apps Script (Code.gs)
 * Mendukung 2-Way Sync:
 * 1. doGet  -> Mengambil saldo & nama pos dari sheet "Ringkasan_Saldo" ke aplikasi web.
 * 2. doPost -> Menyimpan mutasi transaksi & menyusun ulang "Ringkasan_Saldo" secara dinamis.
 */

// ==========================================
// 1. FUNGSI UNTUK MENARIK SALDO KE APLIKASI WEB
// ==========================================
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var summarySheet = ss.getSheetByName("Ringkasan_Saldo");
    
    if (!summarySheet) {
      return responseJSON({ status: "error", message: "Sheet Ringkasan_Saldo belum ada." }, e);
    }

    var data = summarySheet.getDataRange().getValues();
    var tabKeys = [];
    var hutKeys = [];
    var balances = {};
    var currentGroup = ""; // "TABUNGAN" atau "HUTANG"

    for (var i = 0; i < data.length; i++) {
      var colA = String(data[i][0] || "").trim();
      var colB = data[i][1];

      if (!colA) continue;

      var upperA = colA.toUpperCase();
      if (upperA === "TABUNGAN") {
        currentGroup = "TABUNGAN";
        continue;
      } else if (upperA === "HUTANG MODAL" || upperA === "HUTANG") {
        currentGroup = "HUTANG";
        continue;
      } else if (upperA === "TOTAL" || upperA.indexOf("TOTAL ASET") !== -1) {
        currentGroup = "";
        continue;
      }

      var numVal = Number(colB) || 0;
      if (currentGroup === "TABUNGAN") {
        tabKeys.push(colA);
        balances[colA] = numVal;
      } else if (currentGroup === "HUTANG") {
        hutKeys.push(colA);
        balances[colA] = numVal;
      }
    }

    return responseJSON({
      status: "success",
      tabunganKeys: tabKeys,
      hutangKeys: hutKeys,
      balances: balances
    }, e);

  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() }, e);
  }
}

// Helper untuk format JSON / JSONP (menghindari kendala CORS di browser)
function responseJSON(obj, e) {
  var jsonStr = JSON.stringify(obj);
  var callback = (e && e.parameter && e.parameter.callback) ? e.parameter.callback : null;
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + jsonStr + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(jsonStr)
    .setMimeType(ContentService.MimeType.JSON);
}

// ==========================================
// 2. FUNGSI UNTUK MENERIMA DATA DARI APLIKASI WEB
// ==========================================
function doPost(e) {
  try {
    var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    var data = JSON.parse(raw);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // A. Bersihkan sheet Transaksi jika diminta
    if (data.action === "CLEAR_TRANSACTIONS") {
      var txSheet = ss.getSheetByName("Transaksi");
      if (txSheet) {
        txSheet.clear();
        txSheet.appendRow(["ID Transaksi", "Tanggal", "Rekening", "Tipe", "Nominal", "Kategori", "Keterangan", "Saldo Akhir Akun"]);
        txSheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#e2e8f0");
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }

    // B. Catat mutasi transaksi baru jika ada
    if (data.action === "ADD_TRANSACTION" && data.transaction) {
      var txSheet = ss.getSheetByName("Transaksi") || ss.insertSheet("Transaksi");
      if (txSheet.getLastRow() === 0) {
        txSheet.appendRow(["ID Transaksi", "Tanggal", "Rekening", "Tipe", "Nominal", "Kategori", "Keterangan", "Saldo Akhir Akun"]);
        txSheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#e2e8f0");
      }
      var tx = data.transaction;
      txSheet.appendRow([
        tx.id,
        tx.date,
        tx.account,
        tx.type,
        Number(tx.amount || 0),
        tx.category,
        tx.notes,
        Number(tx.resultingBalance || 0)
      ]);
      var lastRow = txSheet.getLastRow();
      txSheet.getRange(lastRow, 5).setNumberFormat("#,##0");
      txSheet.getRange(lastRow, 8).setNumberFormat("#,##0");
    }
    
    // C. Tulis ulang lembar Ringkasan_Saldo dari sel A1 secara rapi & dinamis
    var summarySheet = ss.getSheetByName("Ringkasan_Saldo") || ss.insertSheet("Ringkasan_Saldo");
    summarySheet.clear();
    
    var tabKeys = data.tabunganKeys || [];
    var hutKeys = data.hutangKeys || [];
    var balances = data.balances || {};
    
    var rows = [];
    
    // Bagian 1: Tabungan
    rows.push(["TABUNGAN", ""]);
    var tabStartRow = 2;
    for (var i = 0; i < tabKeys.length; i++) {
      var k = tabKeys[i];
      rows.push([k, Number(balances[k] || 0)]);
    }
    var tabEndRow = rows.length;
    var totalTabRow = tabEndRow + 1;
    rows.push(["TOTAL", 0]);
    
    // Baris kosong pemisah
    rows.push(["", ""]);
    
    // Bagian 2: Hutang Modal
    var hutHeaderRow = rows.length + 1;
    rows.push(["HUTANG MODAL", ""]);
    var hutStartRow = hutHeaderRow + 1;
    for (var j = 0; j < hutKeys.length; j++) {
      var hk = hutKeys[j];
      rows.push([hk, Number(balances[hk] || 0)]);
    }
    var hutEndRow = rows.length;
    var totalHutRow = hutEndRow + 1;
    rows.push(["TOTAL", 0]);
    
    // Baris kosong pemisah
    rows.push(["", ""]);
    
    // Bagian 3: Total Aset Liquid
    var liquidRow = rows.length + 1;
    rows.push(["TOTAL ASET LIQUID", 0]);
    
    // Tulis matriks data ke range A1:B...
    var range = summarySheet.getRange(1, 1, rows.length, 2);
    range.setValues(rows);
    
    // Pasang Formula Google Sheets Resmi
    if (tabEndRow >= tabStartRow) {
      summarySheet.getRange("B" + totalTabRow).setFormula("=SUM(B" + tabStartRow + ":B" + tabEndRow + ")");
    }
    if (hutEndRow >= hutStartRow) {
      summarySheet.getRange("B" + totalHutRow).setFormula("=SUM(B" + hutStartRow + ":B" + hutEndRow + ")");
    }
    summarySheet.getRange("B" + liquidRow).setFormula("=B" + totalTabRow + "+B" + totalHutRow);
    
    // Pewarnaan / Styling Persis Gambar Screenshot:
    summarySheet.getRange("A1:B1").setFontWeight("bold").setBackground("#bbf7d0").setFontColor("#14532d");
    summarySheet.getRange("A" + totalTabRow + ":B" + totalTabRow).setFontWeight("bold").setBackground("#86efac").setFontColor("#14532d");
    summarySheet.getRange("A" + hutHeaderRow + ":B" + hutHeaderRow).setFontWeight("bold").setBackground("#fecdd3").setFontColor("#881337");
    summarySheet.getRange("A" + totalHutRow + ":B" + totalHutRow).setFontWeight("bold").setBackground("#fda4af").setFontColor("#881337");
    summarySheet.getRange("A" + liquidRow + ":B" + liquidRow).setFontWeight("bold").setBackground("#fef08a").setFontColor("#713f12");
    
    summarySheet.getRange("B2:B" + liquidRow).setNumberFormat("#,##0");
    summarySheet.setColumnWidth(1, 220);
    summarySheet.setColumnWidth(2, 160);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
